Q13PyTorch 신경망

import torch, torch.nn as nn
from torch.utils.data import TensorDataset, DataLoader
torch.manual_seed(42); f32 = torch.float32
X_tr = torch.tensor(X_train.values, dtype=f32)   # bool이면 astype 먼저
y_tr = torch.tensor(y_train.values, dtype=f32).view(-1, 1)  # 다중: long
X_va = torch.tensor(X_valid.values, dtype=f32)
y_va = torch.tensor(y_valid.values, dtype=f32).view(-1, 1)
loader = DataLoader(TensorDataset(X_tr, y_tr),
                    batch_size=32, shuffle=True)

model = nn.Sequential(
    nn.Linear(X_tr.shape[1], 64), nn.ReLU(), nn.Dropout(0.2),
    nn.Linear(64, 32), nn.ReLU(), nn.Linear(32, 1))   # ← 표 참고
loss_fn = nn.MSELoss()
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
history = {'loss': [], 'val_loss': []}
best, wait, patience = float('inf'), 0, 5
for epoch in range(50):
    model.train(); total = 0
    for xb, yb in loader:
        optimizer.zero_grad()                  # ① 기울기 초기화
        loss = loss_fn(model(xb), yb)          # ② 예측 + 손실
        loss.backward(); optimizer.step()      # ③ 역전파 ④ 갱신
        total += loss.item() * len(xb)
    model.eval()
    with torch.no_grad():
        val = loss_fn(model(X_va), y_va).item()
    history['loss'].append(total / len(X_tr))
    history['val_loss'].append(val)
    if val < best:
        best, wait = val, 0
        best_state = {k: v.clone() for k, v in model.state_dict().items()}
    else:
        wait += 1
        if wait >= patience: break             # 조기 종료
model.load_state_dict(best_state)
